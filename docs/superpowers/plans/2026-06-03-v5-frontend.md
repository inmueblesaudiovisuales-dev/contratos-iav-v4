function renderAlertas() {
  const bar = document.getElementById('stats-bar');

async function ntBuscarCliente(q) {
  if (q.length < 2) { document.getElementById('nt-clientes-sugeridos').innerHTML = ''; return; }

  const clientes = (data.clientes || []).filter(c =>

async function guardarNuevoTrabajo() {

      inmobiliaria: document.getElementById('nt-inmob-cliente').value.trim()

function toggleFormNota() {
  const f = document.getElementById('eq-nota-form');
  f.style.display = f.style.display === 'none' ? 'block' : 'none';
  if (f.style.display === 'block') document.getElementById('eq-nota-txt').focus();

async function guardarNotaEquipo() {
  const nota = document.getElementById('eq-nota-txt').value.trim();
  if (!nota) return;
  const res = await fetch(API + '/agregarNota', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      trabajoId: DATA?.trabajoId || '',
      clienteId: DATA?.cliente?.id || '',
      descripcion: nota
    })
  });
  const data = await res.json();
  if (!data.ok) return alert('Error: ' + data.error);
  document.getElementById('eq-nota-txt').value = '';
  document.getElementById('eq-nota-form').style.display = 'none';
  // Reload to show new nota
  cargar();
}
